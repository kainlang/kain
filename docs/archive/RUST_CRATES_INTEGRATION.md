### 1. `heck` - Case Mapping Perfection

**Problem:** Unreal is picky about casing. `snake_case` in KAIN must become `PascalCase` in UE5.

**Current Status:** Manual string manipulation with edge case bugs:
```rust
// Current (buggy)
fn to_pascal_case(s: &str) -> String {
    s.split('_')
        .map(|word| {
            let mut chars = word.chars();
            match chars.next() {
                Some(first) => first.to_uppercase().chain(chars).collect(),
                None => String::new(),
            }
        })
        .collect()
}

// Fails on: "gpu_id" -> "GpuId" (should be "GpuID")
// Fails on: "http_request" -> "HttpRequest" (should be "HTTPRequest")
// Fails on: "ui_widget" -> "UiWidget" (should be "UIWidget")
```

**Godmode v3 Solution:**
```rust
use heck::{ToPascalCase, ToSnakeCase, ToKebabCase, ToUpperCamelCase};

// Perfect handling of acronyms and edge cases
"delta_time".to_pascal_case()    // "DeltaTime" ✅
"gpu_id".to_pascal_case()        // "GpuId" ✅ (UE5 convention)
"http_request".to_pascal_case()  // "HttpRequest" ✅
"ui_widget".to_pascal_case()     // "UiWidget" ✅

// Reverse mapping for error messages
"DeltaTime".to_snake_case()      // "delta_time" ✅
```

**Integration Plan:**

1. **Add to Cargo.toml:**
```toml
[dependencies]

```

2. **Update `kain/src/ue5/naming.rs`:**
```rust

/// Convert KAIN identifier to UE5 PascalCase
pub fn to_pascal_case(s: &str) -> String {
    // Handle special UE5 acronyms
    let result = s.to_pascal_case();
    
    // UE5-specific overrides
    match result.as_str() {
        "Gpu" => "GPU".to_string(),
        "Ui" => "UI".to_string(),
        "Http" => "HTTP".to_string(),
        "Ai" => "AI".to_string(),
        "Fps" => "FPS".to_string(),
        "Hud" => "HUD".to_string(),
        _ => result
    }
}

    s.to_snake_case()
}
```

### 3. `clang` - The Engine Scanner (Replaces Python Oracle)

**Problem:** We need to scan UE5 source to auto-generate stdlib metadata.

**Current Plan:** Python script with regex/libclang

**Godmode v3 Solution:** Use Rust `clang` crate directly - **no Python needed!**

```rust
use clang::{Clang, Entity, EntityKind};

/// Scan UE5 source and generate metadata
pub struct Ue5Scanner {
    clang: Clang,
    metadata: EngineMetadata,
}

impl Ue5Scanner {
    pub fn new() -> Self {
        Self {
            clang: Clang::new().unwrap(),
            metadata: EngineMetadata::default(),
        }
    }
    
    /// Scan a UE5 header file
    pub fn scan_header(&mut self, path: &Path) -> Result<()> {
        let index = clang::Index::new(&self.clang, false, false);
        
        let tu = index.parser(path)
            .arguments(&[
                "-x", "c++",
                "-std=c++17",
                "-I", "C:/UE5/Engine/Source/Runtime/Core/Public",
                "-I", "C:/UE5/Engine/Source/Runtime/CoreUObject/Public",
                // ... UE5 include paths
            ])
            .parse()?;
        
        let entity = tu.get_entity();
        self.visit_entity(entity);
        
        Ok(())
    }
    
    fn visit_entity(&mut self, entity: Entity) {
        match entity.get_kind() {
            EntityKind::ClassDecl => self.extract_class(entity),
            EntityKind::FunctionDecl => self.extract_function(entity),
            EntityKind::Method => self.extract_method(entity),
            EntityKind::Namespace => {
                // Recurse into namespace
                for child in entity.get_children() {
                    self.visit_entity(child);
                }
            }
            _ => {}
        }
    }
    
    fn extract_class(&mut self, entity: Entity) {
        // Check for UCLASS macro
        if !self.has_uclass_macro(&entity) {
            return;
        }
        
        let class_name = entity.get_name().unwrap();
        let base_class = entity.get_base_classes()
            .first()
            .and_then(|b| b.get_name());
        
        self.metadata.classes.insert(class_name.clone(), ClassMetadata {
            name: class_name,
            base_class,
            is_actor: entity.get_name().unwrap().starts_with('A'),
            is_component: entity.get_name().unwrap().contains("Component"),
            methods: Vec::new(),
        });
    }
    
    fn extract_method(&mut self, entity: Entity) {
        // Check for UFUNCTION macro
        if !self.has_ufunction_macro(&entity) {
            return;
        }
        
        let func_name = entity.get_name().unwrap();
        let return_type = entity.get_result_type().unwrap().get_display_name();
        
        let parameters = entity.get_arguments()
            .unwrap()
            .iter()
            .map(|arg| Parameter {
                name: arg.get_name().unwrap(),
                param_type: arg.get_type().unwrap().get_display_name(),
            })
            .collect();
        
        self.metadata.functions.insert(func_name.clone(), FunctionMetadata {
            name: func_name,
            return_type,
            parameters,
            is_static: entity.is_static_method(),
            is_const: entity.is_const_method(),
            class_name: entity.get_semantic_parent()
                .and_then(|p| p.get_name()),
        });
    }
    
    fn has_uclass_macro(&self, entity: &Entity) -> bool {
        // Check if entity has UCLASS in its tokens
        entity.get_children()
            .iter()
            .any(|child| {
                child.get_kind() == EntityKind::MacroExpansion &&
                child.get_name().map(|n| n == "UCLASS").unwrap_or(false)
            })
    }
    
    fn has_ufunction_macro(&self, entity: &Entity) -> bool {
        // Check if entity has UFUNCTION in its tokens
        entity.get_children()
            .iter()
            .any(|child| {
                child.get_kind() == EntityKind::MacroExpansion &&
                child.get_name().map(|n| n == "UFUNCTION").unwrap_or(false)
            })
    }
}
```

### 4. `rayon` - Parallel Plugin Factory

**Problem:** Building 1,000 plugins sequentially takes hours.

**Current Status:** Single-threaded compilation

**Godmode v3 Solution:**
```rust
use rayon::prelude::*;

/// Compile multiple plugins in parallel
pub fn compile_plugins_parallel(plugin_dirs: &[PathBuf]) -> Vec<Result<PluginOutput>> {
    plugin_dirs
        .par_iter()  // Parallel iterator
        .map(|dir| {
            // Each plugin compiles on its own thread
            compile_plugin(dir)
        })
        .collect()
}

/// Batch compile entire marketplace catalog
pub fn compile_marketplace_batch(catalog_dir: &Path) -> BatchResult {
    let plugin_dirs: Vec<_> = std::fs::read_dir(catalog_dir)?
        .filter_map(|e| e.ok())
        .map(|e| e.path())
        .filter(|p| p.is_dir())
        .collect();
    
    println!("🚀 Compiling {} plugins across {} cores...", 
        plugin_dirs.len(), 
        rayon::current_num_threads()
    );
    
    let start = std::time::Instant::now();
    
    let results = compile_plugins_parallel(&plugin_dirs);
    
    let elapsed = start.elapsed();
    let success_count = results.iter().filter(|r| r.is_ok()).count();
    
    println!("✅ Compiled {}/{} plugins in {:.2}s", 
        success_count, 
        plugin_dirs.len(),
        elapsed.as_secs_f64()
    );
    
    BatchResult {
        total: plugin_dirs.len(),
        success: success_count,
        failed: results.len() - success_count,
        elapsed,
        results,
    }
}
```

```rust
#[derive(Parser)]
enum Command {
    // ... existing commands
    
    /// Compile multiple plugins in parallel
    #[command(name = "batch")]
    Batch {
        /// Directory containing plugin folders
        #[arg(short, long)]
        catalog: PathBuf,
        
        /// Number of threads (default: all cores)
        #[arg(short, long)]
        threads: Option<usize>,
        
        /// Continue on errors
        #[arg(long)]
        continue_on_error: bool,
    },
}

fn handle_batch(catalog: PathBuf, threads: Option<usize>, continue_on_error: bool) {
    // Set thread pool size
    if let Some(n) = threads {
        rayon::ThreadPoolBuilder::new()
            .num_threads(n)
            .build_global()
            .unwrap();
    }
    
    let result = compile_marketplace_batch(&catalog);
    
    // Report results
    println!("\n📊 Batch Compilation Report:");
    println!("   Total:   {}", result.total);
    println!("   Success: {} ({:.1}%)", result.success, 
        (result.success as f64 / result.total as f64) * 100.0);
    println!("   Failed:  {}", result.failed);
    println!("   Time:    {:.2}s", result.elapsed.as_secs_f64());
    println!("   Speed:   {:.1} plugins/sec", 
        result.total as f64 / result.elapsed.as_secs_f64());
    
    if !continue_on_error && result.failed > 0 {
        std::process::exit(1);
    }
}
```
